cmake_minimum_required(VERSION 3.25)
project(BuildEngineSmokeImGui LANGUAGES CXX)
if(NOT DEFINED IMGUI_PACKAGE_ROOT)
   message(FATAL_ERROR "IMGUI_PACKAGE_ROOT is required")
endif()
add_executable(imgui-consumer main.cpp)
target_compile_features(imgui-consumer PRIVATE cxx_std_23)
target_compile_definitions(imgui-consumer PRIVATE "IMGUI_API=__declspec(dllimport)")
target_include_directories(imgui-consumer PRIVATE "${IMGUI_PACKAGE_ROOT}/include")
target_link_libraries(imgui-consumer PRIVATE "${IMGUI_PACKAGE_ROOT}/lib/win64/Release/libimgui.lib")
