cmake_minimum_required(VERSION 3.25)
project(BuildEngineSmokeLLVMOpenMP LANGUAGES CXX)
if(NOT DEFINED OPENMP_PACKAGE_ROOT OR NOT DEFINED OPENMP_CONFIGURATION)
   message(FATAL_ERROR "Explicit OPENMP_PACKAGE_ROOT and OPENMP_CONFIGURATION required")
endif()
set(_root "${OPENMP_PACKAGE_ROOT}")
set(_cfg "${OPENMP_CONFIGURATION}")
set(_dll "${_root}/bin/win64/${_cfg}/libomp.dll")
set(_implib "${_root}/lib/win64/${_cfg}/libomp.lib")
set(_header "${_root}/include/omp.h")
foreach(_required IN ITEMS "${_dll}" "${_implib}" "${_header}")
   if(NOT EXISTS "${_required}")
      message(FATAL_ERROR "BuildEngine-owned OpenMP artifact missing: ${_required}")
   endif()
endforeach()
add_library(buildengine_openmp SHARED IMPORTED GLOBAL)
set_target_properties(buildengine_openmp PROPERTIES IMPORTED_IMPLIB "${_implib}" IMPORTED_LOCATION "${_dll}" INTERFACE_INCLUDE_DIRECTORIES "${_root}/include")
add_executable(llvm-openmp-consumer main.cpp)
set_property(TARGET llvm-openmp-consumer PROPERTY CXX_SCAN_FOR_MODULES OFF)
target_compile_features(llvm-openmp-consumer PRIVATE cxx_std_23)
target_compile_options(llvm-openmp-consumer PRIVATE -fopenmp)
target_link_libraries(llvm-openmp-consumer PRIVATE buildengine_openmp)
