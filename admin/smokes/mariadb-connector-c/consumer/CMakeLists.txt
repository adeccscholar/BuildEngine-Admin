cmake_minimum_required(VERSION 3.25)
project(BuildEngineMariaDbConsumer LANGUAGES CXX)
if(NOT EXISTS "${MARIADB_INCLUDE_DIR}/mysql.h" OR NOT EXISTS "${MARIADB_LIBRARY}")
   message(FATAL_ERROR "Missing installed MariaDB Connector/C")
endif()
add_executable(mariadb-consumer main.cpp)
set_target_properties(mariadb-consumer PROPERTIES CXX_STANDARD 23 CXX_STANDARD_REQUIRED ON CXX_EXTENSIONS OFF CXX_SCAN_FOR_MODULES OFF)
target_include_directories(mariadb-consumer PRIVATE "${MARIADB_INCLUDE_DIR}")
target_link_libraries(mariadb-consumer PRIVATE "${MARIADB_LIBRARY}")
