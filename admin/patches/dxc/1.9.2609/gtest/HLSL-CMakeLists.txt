# Version-bound BuildEngine port of all original HLSL test translation units.
# TAEF is replaced by original DXC GoogleTest main and assertions.
find_package(DiaSDK REQUIRED)
find_package(D3D12 REQUIRED)

set(LLVM_LINK_COMPONENTS
  support mssupport dxcsupport dxil dxilcontainer dxilrootsignature hlsl
  dxilhash option bitreader bitwriter analysis ipa irreader transformutils)

# Preserve all Windows-specific upstream cases as well as portable cases.
# Neither the original Microsoft TAEF DLL nor the non-Windows omissions
# represent the complete list alone.
add_clang_unittest(ClangHLSLTests
  AllocatorTest.cpp
  CompilerTest.cpp
  DxilContainerTest.cpp
  DxilModuleTest.cpp
  DxilResourceTests.cpp
  DXIsenseTest.cpp
  ExtensionTest.cpp
  FunctionTest.cpp
  HLSLTestOptions.cpp
  LinkerTest.cpp
  MSFileSysTest.cpp
  Objects.cpp
  OptimizerTest.cpp
  OptionsTest.cpp
  PixDiaTest.cpp
  PixTest.cpp
  PixTestUtils.cpp
  RewriterTest.cpp
  SystemValueTest.cpp
  TestMain.cpp
  ValidationTest.cpp
  VerifierTest.cpp)

target_link_libraries(ClangHLSLTests PRIVATE
  dxcompiler HLSLTestLib LLVMDxilContainer LLVMDxilDia gmock
  ${DIASDK_LIBRARIES} ${D3D12_LIBRARIES} shlwapi)
target_include_directories(ClangHLSLTests PRIVATE
  ${DIASDK_INCLUDE_DIRS} ${D3D12_INCLUDE_DIRS}
  ${LLVM_MAIN_INCLUDE_DIR}/dxc/Test ${CLANG_BINARY_DIR}/lib/Sema)
target_compile_definitions(ClangHLSLTests PRIVATE ADECC_DXC_GTEST=1)
add_dependencies(ClangHLSLTests dxcompiler)

# Register all test cases through the original executable's GoogleTest runner.
add_test(NAME DXC.HLSL.GoogleTest
  COMMAND $<TARGET_FILE:ClangHLSLTests>
    --HlslDataDir ${PROJECT_SOURCE_DIR}/tools/clang/test/HLSL)
set_tests_properties(DXC.HLSL.GoogleTest
  PROPERTIES LABELS "BCC64X;DXC;HLSL;original" TIMEOUT 1800)
