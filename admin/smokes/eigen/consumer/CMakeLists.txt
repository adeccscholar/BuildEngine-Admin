cmake_minimum_required(VERSION 3.25)
project(BuildEngineSmokeEigen LANGUAGES CXX)
if(NOT DEFINED EIGEN_PACKAGE_ROOT)
   message(FATAL_ERROR "EIGEN_PACKAGE_ROOT must identify the installed Eigen package")
endif()
add_executable(eigen-consumer main.cpp)
set_property(TARGET eigen-consumer PROPERTY CXX_SCAN_FOR_MODULES OFF)
target_compile_features(eigen-consumer PRIVATE cxx_std_23)
target_include_directories(eigen-consumer PRIVATE "${EIGEN_PACKAGE_ROOT}/include/eigen3")
