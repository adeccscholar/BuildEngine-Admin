cmake_minimum_required(VERSION 3.31)
project(buildengine_openexr_consumer LANGUAGES CXX)
if(NOT DEFINED OPENEXR_PACKAGE_ROOT OR NOT DEFINED IMATH_PACKAGE_ROOT OR NOT DEFINED OPENJPH_PACKAGE_ROOT OR NOT DEFINED LIBDEFLATE_PACKAGE_ROOT OR NOT DEFINED ZSTD_PACKAGE_ROOT)
   message(FATAL_ERROR "OpenEXR, Imath, OpenJPH, libdeflate and zstd package roots are required")
endif()
list(PREPEND CMAKE_PREFIX_PATH "${OPENEXR_PACKAGE_ROOT}" "${IMATH_PACKAGE_ROOT}" "${OPENJPH_PACKAGE_ROOT}" "${LIBDEFLATE_PACKAGE_ROOT}" "${ZSTD_PACKAGE_ROOT}")
set(Imath_DIR "${IMATH_PACKAGE_ROOT}/lib/win64/Release/cmake/Imath")
set(openjph_DIR "${OPENJPH_PACKAGE_ROOT}/lib/win64/Release/cmake/openjph")
set(libdeflate_DIR "${LIBDEFLATE_PACKAGE_ROOT}/lib/win64/Release/cmake/libdeflate")
set(zstd_DIR "${ZSTD_PACKAGE_ROOT}/lib/win64/Release/cmake/zstd")
set(OpenEXR_DIR "${OPENEXR_PACKAGE_ROOT}/lib/win64/Release/cmake/OpenEXR")
if(NOT EXISTS "${OpenEXR_DIR}/OpenEXRConfig.cmake")
   message(FATAL_ERROR "Installed OpenEXR CMake package is missing: ${OpenEXR_DIR}/OpenEXRConfig.cmake")
endif()
find_package(Imath CONFIG REQUIRED)
find_package(Threads REQUIRED)
find_package(openjph CONFIG REQUIRED)
find_package(libdeflate CONFIG REQUIRED)
find_package(zstd CONFIG REQUIRED)
find_package(OpenEXR CONFIG REQUIRED)
add_executable(openexr-consumer main.cpp)
target_compile_features(openexr-consumer PRIVATE cxx_std_23)
target_link_libraries(openexr-consumer PRIVATE OpenEXR::OpenEXR)
