cmake_minimum_required(VERSION 3.25)
project(AbseilConsumer LANGUAGES CXX)
find_package(absl CONFIG REQUIRED)
if(NOT TARGET absl::abseil_dll)
   message(FATAL_ERROR "Abseil package does not export absl::abseil_dll")
endif()
get_target_property(_abseil_dll_type absl::abseil_dll TYPE)
if(NOT _abseil_dll_type STREQUAL "SHARED_LIBRARY")
   message(FATAL_ERROR "Abseil consumer requires a DLL package, got ${_abseil_dll_type}")
endif()
add_executable(abseil-consumer main.cpp)
set_property(TARGET abseil-consumer PROPERTY CXX_SCAN_FOR_MODULES OFF)
target_compile_features(abseil-consumer PRIVATE cxx_std_23)
target_link_libraries(abseil-consumer PRIVATE absl::strings)
