cmake_minimum_required(VERSION 3.25)
project(BuildEngineTesseractOCRSmoke LANGUAGES CXX)
find_package(Leptonica 1.87 CONFIG REQUIRED)
find_package(Tesseract 5.5 CONFIG REQUIRED)
if(NOT DEFINED OCR_TESSDATA_DIRECTORY OR NOT EXISTS "${OCR_TESSDATA_DIRECTORY}/eng.traineddata")
   message(FATAL_ERROR "OCR_TESSDATA_DIRECTORY must contain verified eng.traineddata")
endif()
if(NOT Tesseract_INCLUDE_DIRS OR NOT EXISTS "${Tesseract_INCLUDE_DIRS}/tesseract/baseapi.h")
   message(FATAL_ERROR "Installed Tesseract include path unavailable")
endif()
configure_file(ocr_paths.h.in "${CMAKE_CURRENT_BINARY_DIR}/ocr_paths.h" @ONLY)
add_executable(tesseract-ocr-consumer main.cpp)
set_property(TARGET tesseract-ocr-consumer PROPERTY CXX_SCAN_FOR_MODULES OFF)
target_compile_features(tesseract-ocr-consumer PRIVATE cxx_std_23)
target_include_directories(tesseract-ocr-consumer PRIVATE "${Tesseract_INCLUDE_DIRS}" "${Leptonica_INCLUDE_DIRS}" "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(tesseract-ocr-consumer PRIVATE Tesseract::libtesseract)
