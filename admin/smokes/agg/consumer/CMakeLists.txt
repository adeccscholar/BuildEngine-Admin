cmake_minimum_required(VERSION 3.25)
project(AggConsumer LANGUAGES CXX)
find_package(agg CONFIG REQUIRED)
add_executable(agg-consumer main.cpp)
set_property(TARGET agg-consumer PROPERTY CXX_SCAN_FOR_MODULES OFF)
target_compile_features(agg-consumer PRIVATE cxx_std_23)
target_link_libraries(agg-consumer PRIVATE agg::agg)
