cmake_minimum_required(VERSION 3.31)
project(buildengine_ceres_consumer LANGUAGES CXX)
if(NOT DEFINED CERES_PACKAGE_ROOT OR NOT DEFINED EIGEN_PACKAGE_ROOT OR NOT DEFINED GLOG_PACKAGE_ROOT OR NOT DEFINED GFLAGS_PACKAGE_ROOT)
   message(FATAL_ERROR "Ceres, Eigen, glog and gflags package roots are required")
endif()
list(PREPEND CMAKE_PREFIX_PATH "${CERES_PACKAGE_ROOT}" "${GLOG_PACKAGE_ROOT}" "${GFLAGS_PACKAGE_ROOT}")
set(Ceres_DIR "${CERES_PACKAGE_ROOT}/lib/win64/Release/cmake/Ceres")
set(Eigen3_DIR "${EIGEN_PACKAGE_ROOT}/lib/cmake/Eigen3")
set(glog_DIR "${GLOG_PACKAGE_ROOT}/lib/win64/Release/cmake/glog")
set(gflags_DIR "${GFLAGS_PACKAGE_ROOT}/lib/win64/Release/cmake/gflags")
set(gflags_USE_TARGET_NAMESPACE TRUE)
find_package(Ceres CONFIG REQUIRED)
add_executable(ceres-consumer main.cpp)
target_compile_features(ceres-consumer PRIVATE cxx_std_23)
target_include_directories(ceres-consumer PRIVATE "${EIGEN_PACKAGE_ROOT}/include/eigen3")
target_link_libraries(ceres-consumer PRIVATE Ceres::ceres)
