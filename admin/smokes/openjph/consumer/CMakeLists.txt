cmake_minimum_required(VERSION 3.31)
project(buildengine_openjph_consumer LANGUAGES CXX)
if(NOT DEFINED OPENJPH_PACKAGE_ROOT)
   message(FATAL_ERROR "OPENJPH_PACKAGE_ROOT is required")
endif()
list(PREPEND CMAKE_PREFIX_PATH "${OPENJPH_PACKAGE_ROOT}")
find_package(openjph CONFIG REQUIRED)
add_executable(openjph-consumer main.cpp)
target_compile_features(openjph-consumer PRIVATE cxx_std_23)
target_link_libraries(openjph-consumer PRIVATE openjph)
