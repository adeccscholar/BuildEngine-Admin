cmake_minimum_required(VERSION 3.31)
project(buildengine_glog_consumer LANGUAGES CXX)
if(NOT DEFINED GLOG_PACKAGE_ROOT OR NOT DEFINED GFLAGS_PACKAGE_ROOT)
   message(FATAL_ERROR "GLOG_PACKAGE_ROOT and GFLAGS_PACKAGE_ROOT are required")
endif()
list(PREPEND CMAKE_PREFIX_PATH "${GFLAGS_PACKAGE_ROOT}" "${GLOG_PACKAGE_ROOT}")
set(gflags_DIR "${GFLAGS_PACKAGE_ROOT}/lib/win64/Release/cmake/gflags")
set(glog_DIR "${GLOG_PACKAGE_ROOT}/lib/win64/Release/cmake/glog")
set(GFLAGS_USE_TARGET_NAMESPACE TRUE)
find_package(gflags CONFIG REQUIRED)
find_package(glog CONFIG REQUIRED)
# Upstream glog with gflags enabled exposes gflags as a PUBLIC dependency.
# Never hide a broken glog package export by adding a consumer-only link.
if(NOT TARGET gflags::gflags)
   message(FATAL_ERROR "Missing imported target gflags::gflags")
endif()
get_target_property(lstGlogDependencies glog::glog INTERFACE_LINK_LIBRARIES)
if(NOT lstGlogDependencies OR NOT "gflags::gflags" IN_LIST lstGlogDependencies)
   message(FATAL_ERROR "glog::glog must propagate gflags::gflags, got: ${lstGlogDependencies}")
endif()
get_target_property(lstGflagsIncludeDirectories gflags::gflags INTERFACE_INCLUDE_DIRECTORIES)
if(NOT lstGflagsIncludeDirectories)
   message(FATAL_ERROR "gflags::gflags has no transitive include directories")
endif()
add_executable(glog-consumer main.cpp)
target_compile_features(glog-consumer PRIVATE cxx_std_23)
target_link_libraries(glog-consumer PRIVATE glog::glog)
