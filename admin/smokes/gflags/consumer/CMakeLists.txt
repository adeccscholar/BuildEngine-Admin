cmake_minimum_required(VERSION 3.31)
project(buildengine_gflags_consumer LANGUAGES CXX)
if(NOT DEFINED GFLAGS_PACKAGE_ROOT)
   message(FATAL_ERROR "GFLAGS_PACKAGE_ROOT is required")
endif()
set(gflags_USE_TARGET_NAMESPACE TRUE)
find_package(gflags CONFIG REQUIRED PATHS "${GFLAGS_PACKAGE_ROOT}/lib/win64/Release/cmake/gflags" "${GFLAGS_PACKAGE_ROOT}/lib/cmake/gflags" NO_DEFAULT_PATH)
add_executable(gflags-consumer main.cpp)
target_compile_features(gflags-consumer PRIVATE cxx_std_23)
target_link_libraries(gflags-consumer PRIVATE gflags::gflags)
