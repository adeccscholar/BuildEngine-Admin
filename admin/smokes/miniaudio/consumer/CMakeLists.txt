cmake_minimum_required(VERSION 3.31)
project(buildengine_miniaudio_consumer LANGUAGES CXX)
if(NOT DEFINED MINIAUDIO_PACKAGE_ROOT)
   message(FATAL_ERROR "MINIAUDIO_PACKAGE_ROOT must point to the installed BuildEngine package")
endif()
add_executable(miniaudio-consumer main.cpp)
target_compile_features(miniaudio-consumer PRIVATE cxx_std_23)
target_include_directories(miniaudio-consumer PRIVATE "${MINIAUDIO_PACKAGE_ROOT}/include")
target_compile_definitions(miniaudio-consumer PRIVATE MA_NO_DECODING MA_NO_ENCODING)
if(WIN32)
   target_link_libraries(miniaudio-consumer PRIVATE winmm ole32)
endif()
