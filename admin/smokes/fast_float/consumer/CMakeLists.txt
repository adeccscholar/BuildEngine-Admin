cmake_minimum_required(VERSION 3.25)
project(FastFloatConsumer LANGUAGES CXX)
find_package(FastFloat CONFIG REQUIRED)
add_executable(fast_float-consumer main.cpp)
set_property(TARGET fast_float-consumer PROPERTY CXX_SCAN_FOR_MODULES OFF)
target_compile_features(fast_float-consumer PRIVATE cxx_std_23)
target_link_libraries(fast_float-consumer PRIVATE FastFloat::fast_float)
