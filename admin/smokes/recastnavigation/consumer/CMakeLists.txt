cmake_minimum_required(VERSION 3.31)
project(buildengine_recast_consumer LANGUAGES CXX)
if(NOT DEFINED RECAST_PACKAGE_ROOT)
   message(FATAL_ERROR "RECAST_PACKAGE_ROOT is required")
endif()
add_executable(recast-consumer main.cpp)
target_compile_features(recast-consumer PRIVATE cxx_std_23)
target_include_directories(recast-consumer PRIVATE "${RECAST_PACKAGE_ROOT}/include/recastnavigation")
find_library(RECAST_LIBRARY NAMES Recast libRecast PATHS "${RECAST_PACKAGE_ROOT}/lib/win64/Release" NO_DEFAULT_PATH REQUIRED)
find_library(DETOUR_LIBRARY NAMES Detour libDetour PATHS "${RECAST_PACKAGE_ROOT}/lib/win64/Release" NO_DEFAULT_PATH REQUIRED)
target_link_libraries(recast-consumer PRIVATE "${RECAST_LIBRARY}" "${DETOUR_LIBRARY}")
