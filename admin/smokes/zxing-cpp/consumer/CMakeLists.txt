cmake_minimum_required(VERSION 3.31)
project(buildengine_zxing_consumer LANGUAGES CXX)
if(NOT DEFINED ZXING_PACKAGE_ROOT)
   message(FATAL_ERROR "ZXING_PACKAGE_ROOT is required")
endif()
add_executable(zxing-consumer main.cpp)
target_compile_features(zxing-consumer PRIVATE cxx_std_23)
target_include_directories(zxing-consumer PRIVATE "${ZXING_PACKAGE_ROOT}/include")
find_library(ZXING_LIBRARY NAMES ZXing libZXing PATHS "${ZXING_PACKAGE_ROOT}/lib/win64/Release" NO_DEFAULT_PATH REQUIRED)
target_link_libraries(zxing-consumer PRIVATE "${ZXING_LIBRARY}")
