cmake_minimum_required(VERSION 3.31)
project(buildengine_imath_consumer LANGUAGES CXX)
if(NOT DEFINED IMATH_PACKAGE_ROOT)
   message(FATAL_ERROR "IMATH_PACKAGE_ROOT is required")
endif()
list(PREPEND CMAKE_PREFIX_PATH "${IMATH_PACKAGE_ROOT}")
find_package(Imath CONFIG REQUIRED)
add_executable(imath-consumer main.cpp)
target_compile_features(imath-consumer PRIVATE cxx_std_23)
target_link_libraries(imath-consumer PRIVATE Imath::Imath)
