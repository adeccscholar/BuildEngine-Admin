cmake_minimum_required(VERSION 3.25)
project(BuildEngineSmokeGoogleBenchmark LANGUAGES CXX)
find_package(benchmark CONFIG REQUIRED)
add_executable(google-benchmark-consumer main.cpp)
set_property(TARGET google-benchmark-consumer PROPERTY CXX_SCAN_FOR_MODULES OFF)
target_compile_features(google-benchmark-consumer PRIVATE cxx_std_23)
target_link_libraries(google-benchmark-consumer PRIVATE benchmark::benchmark)
