function(RequireSharedTarget theTarget)
   if(NOT TARGET ${theTarget})
      message(FATAL_ERROR "Boost gate ${BOOST_GATE}: missing target ${theTarget}")
   endif()
   get_target_property(theType ${theTarget} TYPE)
   if(NOT theType STREQUAL "SHARED_LIBRARY")
      message(FATAL_ERROR "Boost gate ${BOOST_GATE}: ${theTarget} must be SHARED_LIBRARY, got '${theType}'")
   endif()

   get_target_property(theConfigurations ${theTarget} IMPORTED_CONFIGURATIONS)
   set(bLocationFound FALSE)
   set(bImportLibraryFound FALSE)
   foreach(theConfiguration IN LISTS theConfigurations)
      string(TOUPPER "${theConfiguration}" theUpperConfiguration)
      get_target_property(theLocation ${theTarget} "IMPORTED_LOCATION_${theUpperConfiguration}")
      get_target_property(theImportLibrary ${theTarget} "IMPORTED_IMPLIB_${theUpperConfiguration}")
      if(theLocation AND EXISTS "${theLocation}")
         set(bLocationFound TRUE)
      endif()
      if(theImportLibrary AND EXISTS "${theImportLibrary}")
         set(bImportLibraryFound TRUE)
      endif()
   endforeach()
   if(NOT bLocationFound)
      get_target_property(theLocation ${theTarget} IMPORTED_LOCATION)
      if(theLocation AND EXISTS "${theLocation}")
         set(bLocationFound TRUE)
      endif()
   endif()
   if(NOT bImportLibraryFound)
      get_target_property(theImportLibrary ${theTarget} IMPORTED_IMPLIB)
      if(theImportLibrary AND EXISTS "${theImportLibrary}")
         set(bImportLibraryFound TRUE)
      endif()
   endif()
   if(NOT bLocationFound OR NOT bImportLibraryFound)
      message(FATAL_ERROR "Boost gate ${BOOST_GATE}: ${theTarget} does not expose an existing DLL/import-library pair")
   endif()
endfunction()

function(RequireInterfaceTarget theTarget)
   if(NOT TARGET ${theTarget})
      message(FATAL_ERROR "Boost gate ${BOOST_GATE}: missing target ${theTarget}")
   endif()
   get_target_property(theType ${theTarget} TYPE)
   if(NOT theType STREQUAL "INTERFACE_LIBRARY")
      message(FATAL_ERROR "Boost gate ${BOOST_GATE}: ${theTarget} must be INTERFACE_LIBRARY, got '${theType}'")
   endif()
endfunction()

function(AddEvidenceRuntime theTarget theRuntimeCase)
   add_executable(${theTarget} runtime.cpp)
   set_property(TARGET ${theTarget} PROPERTY CXX_SCAN_FOR_MODULES OFF)
   target_compile_features(${theTarget} PRIVATE cxx_std_23)
   target_compile_definitions(${theTarget} PRIVATE
      ADECC_BOOST_GATE_CODE=${iGateCode}
      ADECC_BOOST_RUNTIME_CASE=${theRuntimeCase}
      ADECC_BOOST_GATE_ID="${BOOST_GATE}")
   target_link_libraries(${theTarget} PRIVATE ${ARGN})
endfunction()
