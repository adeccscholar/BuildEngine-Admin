cmake_minimum_required(VERSION 3.25)
project(AdeccSQLiteDirectBCC64X NONE)
include(GNUInstallDirs)
if(NOT DEFINED SQLITE_SOURCE_DIR OR NOT DEFINED BCC64X_EXECUTABLE)
  message(FATAL_ERROR "SQLITE_SOURCE_DIR and BCC64X_EXECUTABLE are required")
endif()
if(CMAKE_BUILD_TYPE STREQUAL "Debug")
  set(_base sqlite3d)
  set(_opt -O0)
else()
  set(_base sqlite3)
  set(_opt -O2)
endif()
set(_obj "${CMAKE_CURRENT_BINARY_DIR}/sqlite3.obj")
set(_dll "${CMAKE_CURRENT_BINARY_DIR}/${_base}.dll")
set(_lib "${CMAKE_CURRENT_BINARY_DIR}/${_base}.lib")
add_custom_command(OUTPUT "${_dll}" "${_lib}"
  COMMAND "${BCC64X_EXECUTABLE}" -q ${_opt} $<$<NOT:$<CONFIG:Debug>>:-DNDEBUG> -DSQLITE_THREADSAFE=1 "-DSQLITE_API=__declspec(dllexport)" -c "${SQLITE_SOURCE_DIR}/sqlite3.c" -o "${_obj}"
  COMMAND "${BCC64X_EXECUTABLE}" -q --rsp-quoting=windows -tD -o "${_dll}" "${_obj}" "-Wl,--out-implib,${_lib}"
  DEPENDS "${SQLITE_SOURCE_DIR}/sqlite3.c" VERBATIM)
add_custom_target(sqlite3-direct ALL DEPENDS "${_dll}" "${_lib}")
install(FILES "${_dll}" DESTINATION "${CMAKE_INSTALL_BINDIR}")
install(FILES "${_lib}" DESTINATION "${CMAKE_INSTALL_LIBDIR}")
configure_file("${CMAKE_CURRENT_SOURCE_DIR}/SQLite3Config.cmake.in" "${CMAKE_CURRENT_BINARY_DIR}/SQLite3Config.cmake" @ONLY)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/SQLite3Config.cmake" DESTINATION "${CMAKE_INSTALL_LIBDIR}/cmake/SQLite3")
