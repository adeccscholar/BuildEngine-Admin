cmake_minimum_required(VERSION 3.25)
project(AdeccGLEW VERSION 2.3.1 LANGUAGES C)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
if(NOT DEFINED GLEW_SOURCE_DIR OR NOT EXISTS "${GLEW_SOURCE_DIR}/src/glew.c")
  message(FATAL_ERROR "GLEW_SOURCE_DIR must reference the official GLEW 2.3.1 release tree")
endif()
if(NOT DEFINED OPENGL_LIBRARY OR NOT EXISTS "${OPENGL_LIBRARY}")
  message(FATAL_ERROR "OPENGL_LIBRARY is required")
endif()
set(_src "${GLEW_SOURCE_DIR}/src/glew.c")
file(READ "${_src}" _text)
set(_guard "#if defined(_WIN32) && defined(GLEW_BUILD) && defined(__GNUC__)")
set(_bccguard "#if defined(_WIN32) && defined(GLEW_BUILD) && defined(__GNUC__) && !(defined(__BORLANDC__) && defined(_WIN64) && defined(__MINGW64__))")
string(FIND "${_text}" "${_guard}" _pos)
if(_pos EQUAL -1)
  message(FATAL_ERROR "GLEW DLL startup guard not found; refusing to patch blindly")
endif()
string(REPLACE "${_guard}" "${_bccguard}" _text "${_text}")
set(_patched "${CMAKE_CURRENT_BINARY_DIR}/glew-bcc64x.c")
file(WRITE "${_patched}" "${_text}")
add_library(GLEW SHARED "${_patched}")
add_library(GLEW::GLEW ALIAS GLEW)
target_compile_options(GLEW PRIVATE -include stddef.h)
target_include_directories(GLEW PUBLIC $<BUILD_INTERFACE:${GLEW_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_compile_definitions(GLEW PRIVATE GLEW_BUILD)
target_link_libraries(GLEW PRIVATE "${OPENGL_LIBRARY}")
set_target_properties(GLEW PROPERTIES OUTPUT_NAME glew32 DEBUG_POSTFIX d PREFIX "" IMPORT_PREFIX "")
install(TARGETS GLEW EXPORT GLEWTargets RUNTIME DESTINATION "${CMAKE_INSTALL_BINDIR}" LIBRARY DESTINATION "${CMAKE_INSTALL_LIBDIR}" ARCHIVE DESTINATION "${CMAKE_INSTALL_LIBDIR}")
write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/GLEWConfigVersion.cmake" VERSION 2.3.1 COMPATIBILITY SameMajorVersion)
configure_package_config_file("${CMAKE_CURRENT_SOURCE_DIR}/GLEWConfig.cmake.in" "${CMAKE_CURRENT_BINARY_DIR}/GLEWConfig.cmake" INSTALL_DESTINATION "${CMAKE_INSTALL_LIBDIR}/cmake/GLEW")
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/GLEWConfig.cmake" "${CMAKE_CURRENT_BINARY_DIR}/GLEWConfigVersion.cmake" DESTINATION "${CMAKE_INSTALL_LIBDIR}/cmake/GLEW")
install(EXPORT GLEWTargets FILE GLEWTargets.cmake NAMESPACE GLEW:: DESTINATION "${CMAKE_INSTALL_LIBDIR}/cmake/GLEW")
